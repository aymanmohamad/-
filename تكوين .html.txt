function dateToSerial(s){ if(!s) return null; const p=s.split('-').map(Number); return Math.round((Date.UTC(p[0],p[1]-1,p[2])-Date.UTC(1899,11,30))/86400000); }
function sheetFromAoA(aoa){
  const ws = XLSX.utils.aoa_to_sheet(aoa);
  ws['!cols']=[{wch:8.4},{wch:17},{wch:18},{wch:14},{wch:38.3},{wch:40},{wch:31.4},{wch:25.7},{wch:24.9}];
  for(let R=6; R<aoa.length; R++){
    const addr = XLSX.utils.encode_cell({r:R, c:8});
    const cell = ws[addr];
    if(cell && cell.v!=null && cell.v!==''){
      const ser = dateToSerial(String(cell.v));
      if(ser!==null){ cell.t='n'; cell.v=ser; cell.z='mm-dd-yy'; }
    }
  }
  return ws;
}
function downloadWorkbook(aoa, sheetName, filename){
  if(typeof XLSX!=='undefined'){
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, sheetFromAoA(aoa), sheetName);
    XLSX.writeFile(wb, filename);
    return 'xlsx';
  }
  const csv = '\ufeff' + aoa.map(row=>row.map(v=>{
    const s=(v==null?'':String(v));
    return /[",\n]/.test(s) ? '"'+s.replace(/"/g,'""')+'"' : s;
  }).join(',')).join('\r\n');
  const blob = new Blob([csv], {type:'text/csv;charset=utf-8'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(blob); a.download=filename.replace(/\.xlsx$/,'')+'.csv';
  document.body.appendChild(a); a.click(); a.remove();
  return 'csv';
}

              if(emailV && prev.email && emailV!==prev.email) conflicts++;
              map[seq]={email:emailV||prev.email, fix:fixV||prev.fix};
            } else {
              map[seq]={email:emailV, fix:fixV};
            }
          }
        });
        if(got) filesOk++; else filesBad++;
      }catch(err){ filesBad++; }

      }
    };
    reader.onerror=()=>{ filesBad++; if(--pending===0){ mStatus('خطأ في قراءة الملفات.','err'); } };
    reader.readAsArrayBuffer(f);
  });
